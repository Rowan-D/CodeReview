cmake_minimum_required(VERSION 3.20)
project(CodeReview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Concurrent)
add_library(documents src/document.cpp src/gitdiff.cpp src/comparison.cpp src/syntax.cpp)
target_link_libraries(documents PUBLIC Qt6::Core)
target_include_directories(documents PUBLIC src)
add_executable(code-review src/main.cpp src/canvas.cpp)
target_link_libraries(code-review PRIVATE documents Qt6::Widgets Qt6::Concurrent)
target_compile_options(code-review PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
  add_executable(viewer-tests tests/tests.cpp src/canvas.cpp)
  target_link_libraries(viewer-tests PRIVATE documents Qt6::Widgets Qt6::Concurrent Qt6::Test)
  add_test(NAME viewer-tests COMMAND viewer-tests)
  set_tests_properties(viewer-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS code-review RUNTIME DESTINATION bin)
